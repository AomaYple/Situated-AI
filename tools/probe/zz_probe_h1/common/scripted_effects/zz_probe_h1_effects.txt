	every_country = {
		if = {
			limit = { NOT = { has_variable = sitai_probe_dose_assigned } }
			set_variable = { name = sitai_probe_dose_assigned value = 1 }
			random_list = {
				25 = { set_variable = { name = sitai_probe_dose_low value = 1 } }
				25 = { set_variable = { name = sitai_probe_dose_mid value = 1 } }
				25 = { set_variable = { name = sitai_probe_dose_high value = 1 } }
				25 = { }
			}
		}
	}
}
