natural_disaster_events.1 = { #Small Earthquake

	placement = scope:earthquake_state

	desc = natural_disaster_events.1.d

	event_image = {
		video = "unspecific_devastation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		any_scope_state = {
			state_region = {

		}

		random_scope_state = {
			limit = {
				state_region = {

				}
			}
			save_scope_as = earthquake_state
			state_region = {
				set_variable = {
					name = natural_disaster_cooldown
					days = normal_modifier_time
				}
			}
		}
	}

	option = { #No extra help
		name = natural_disaster_events.1.a
		default_option = yes
		scope:earthquake_state.state_region = {
			add_devastation = 20
		}

		# SITAI BEGIN event:natural_disaster_events.1.a
		# ACTION event_disaster_1_a [political];
		ai_chance = {
			base = 0
			# Intentional AI-only policy; omitted native default is unknown.
			modifier = { trigger = { is_ai = yes } add = 50 }
		modifier = { trigger = { is_ai = yes exists = scope:earthquake_state scope:earthquake_state = { state_population >= 1 } } add = -10 }
		modifier = { trigger = { is_ai = yes exists = scope:earthquake_state scope:earthquake_state = { state_population >= 1 } scope:earthquake_state = { state_population >= 100000 } } add = -4 }
		}
		# SITAI END event:natural_disaster_events.1.a
		}

	option = { #Extra help
		name = natural_disaster_events.1.b
		add_modifier = {
			name = modifier_earthquake_relief
			multiplier = money_amount_multiplier_medium
			days = short_modifier_time
		}
		scope:earthquake_state.state_region = {
			add_devastation = 10
		}

		# SITAI BEGIN event:natural_disaster_events.1.b
		# ACTION event_disaster_1_b [political];
		ai_chance = {
			base = 0
			# Intentional AI-only policy; omitted native default is unknown.
			modifier = { trigger = { is_ai = yes } add = 50 }
		modifier = { trigger = { is_ai = yes exists = scope:earthquake_state scope:earthquake_state = { state_population >= 1 } } add = 10 }
		modifier = { trigger = { is_ai = yes exists = scope:earthquake_state scope:earthquake_state = { state_population >= 1 } scope:earthquake_state = { state_population >= 100000 } } add = 4 }
		modifier = { trigger = { is_ai = yes exists = scope:earthquake_state scope:earthquake_state = { state_population >= 1 } } add = -8 }
		modifier = { trigger = { is_ai = yes exists = scope:earthquake_state scope:earthquake_state = { state_population >= 1 } in_default = no } add = 4 }

			state_region = {

			limit = { exists = var:number_of_mega_floods }

			}
