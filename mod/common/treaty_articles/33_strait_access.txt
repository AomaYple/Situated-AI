strait_access = {
    kind = directed
    cost = 10

    relations_progress_per_day = 0.5
    relations_improvement_max = 20

	max_target_involvement = 250
	target_involvement_applies_to = source_country

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/strait_access_treaties.dds"

	flags = {
		is_strait_access
		can_be_enforced
		ai_consider_threaten_naval_hostilities
		can_be_renegotiated
		giftable
		friendly
	}

	maintenance_paid_by = target_country
	usage_limit = once_per_side

    unlocked_by_technologies = {
        international_relations
    }

	visible = {
		has_any_strait_province = yes
	}

	possible = {
		has_any_strait_province = yes
		has_any_strait_control = yes
	}

    can_ratify = {
        custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
            NOT = {
                scope:source_country = {
                    any_scope_treaty = {
						binds = scope:target_country
                        hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							has_type = strait_access
							source_country = scope:source_country
						}
                    }
                }
            }
        }
        custom_tooltip = {
            text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
            NOT = {
				scope:treaty = {
					any_scope_article_option = {
						has_type = no_strait_closure
						source_country = scope:source_country
					}
				}
            }
        }
    }

    on_entry_into_force = {}

	on_withdrawal = {}
	on_break = {
		scope:withdrawing_country = {
			change_relations = { country = scope:non_withdrawing_country value = -10 }
		}
	}

	ai = {

		treaty_categories = {
			military
		}

		evaluation_chance = {
			value = 0

			if = {
				limit = {
					country_rank >= rank_value:unrecognized_major_power
				}
				add = {
					value = country_rank
					multiply = 0.025
				}
			}

			# Any country a strait controller is hurting looks for relief, whatever its rank
			add = {
				value = worst_strait_access_grievance
				multiply = 0.005
			}
		}

		article_ai_usage = {
			request
			offer
		}

		proposal_weight = {
			value = 1

			# Pursued harder the more the strait controller is hurting us
			add = {
				value = "strait_access_grievance_against(scope:source_country)"
				multiply = 0.04
			}

			if = {
				limit = {
					root = scope:target_country
					has_law_or_variant = law_type:law_free_trade # Free traders ask for no strait closure for anyone instead
				}
				value = 0
			}

		# SITAI BEGIN information:strait_access.ai.proposal_weight
		# ACTION program_8c0c0ccbfb29c4fb [joint]; signed-envelope consumer action
		if = { limit = { exists = ROOT ROOT = { is_ai = yes } } multiply = {
			value = 1
			# GROUP program_8c0c0ccbfb29c4fb_treaty_influence_pressure_negative
			add = { value = 0
				# treaty_strait_access_capacity_deficit [diplomacy]; ROOT影响力已为负，升档同一外交容量压力；与低于占用档去重，不把零余额或未知当赤字
				max = { value = 0 if = { limit = { exists = ROOT ROOT = { is_ai = yes OR = { AND = { OR = { influence >= 0 NOT = { influence >= 0 } } } AND = { total_population >= 0 } } } } add = { value = 0 if = { limit = { OR = { AND = { exists = ROOT ROOT = { NOT = { influence < 0.0 } } } } }  } else = { add = -0.2 } } } }
				# treaty_strait_access_capacity [diplomacy]; 原版每方影响力占用10；ROOT当前余额低于本条款精确占用，仅计边际外交容量压力，不增加资格门
				max = { value = 0 if = { limit = { exists = ROOT ROOT = { is_ai = yes OR = { AND = { OR = { influence >= 0 NOT = { influence >= 0 } } } AND = { total_population >= 0 } } } } add = { value = 0 if = { limit = { OR = { AND = { exists = ROOT ROOT = { NOT = { influence < 10.0 } } } } }  } else = { add = -0.1 } } } }
			}
			min = 0.8
			max = 1
		} }
		# SITAI END information:strait_access.ai.proposal_weight
	}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					scope:source_country ?= this
				}

				# We are doing this to be able to reuse loc across treaties and diplomatic actions
				scope:source_country = {
					save_temporary_scope_as = target
				}
				scope:target_country = {
					save_temporary_scope_as = actor
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -10
				}

				add = {
					desc = "concept_strait_controller"
					value = { # A combination of both dependencies
						add = {
							value = {
								scope:target_country = {
									add = "strait_dependence_by(scope:source_country)"
								}
							}
							multiply = 0.5
						}
						add = {
							value = "strait_dependence_by(scope:target_country)"
							multiply = 0.5
						}
					}
					multiply = -0.15
					min = -15
				}

				if = {
					limit = {
						"num_alliances_and_defensive_pacts_with_rivals(scope:target_country)" > 0
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_ALLIED_TO_RIVALS"
						value = "num_alliances_and_defensive_pacts_with_rivals(scope:target_country)"
						multiply = -15
					}
					min = -20
				}

				if = {
					limit = {
						"num_alliances_and_defensive_pacts_with_allies(scope:target_country)" > 0
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_ALLIED_TO_ALLIES"
						value = "num_alliances_and_defensive_pacts_with_allies(scope:target_country)"
						multiply = 10
					}
					max = 20
				}

				if = {
					limit = {
						"num_shared_rivals(scope:target_country)" > 0
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_SHARED_RIVALS"
						value = "num_shared_rivals(scope:target_country)"
						multiply = 10
						max = 20
					}
				}

				if = {
					limit = {
						exists = power_bloc
						exists = scope:target_country.power_bloc
					}
					if = {
						limit = {
							power_bloc = scope:target_country.power_bloc
						}
						add = {
							desc = "DIPLOMATIC_ACCEPTANCE_SAME_POWER_BLOC"
							value = 10
						}
					}
					else = {
						add = {
							desc = "DIPLOMATIC_ACCEPTANCE_DIFFERENT_POWER_BLOC"
							value = -25
						}
					}
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = antagonistic
								}
								has_attitude = {
									who = scope:target_country
									attitude = domineering
								}
								has_attitude = {
									who = scope:target_country
									attitude = belligerent
								}
								has_attitude = {
									who = scope:target_country
									attitude = rebellious
								}
							}
						}
						value = -1000
					}
				}
			}
			else = {
				scope:source_country = {
					save_temporary_scope_as = actor
				}
				scope:target_country = {
					save_temporary_scope_as = target
				}

				if = {
					limit = {
						scope:target_country = {
							"strait_dependence_by(scope:source_country)" < define:NDiplomacy|STRAIT_GRIEVANCE_MIN_DEPENDENCE
						}
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_NO_STRAIT_DEPENDENCE"
						value = -10
					}
				}
				else = {
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_STRAIT_DEPENDENCE"
						value = {
							scope:target_country = {
								add = "strait_dependence_by(scope:source_country)"
							}
						}
						multiply = 0.3
						max = 25
					}
				}
			}
		}

		wargoal_score_multiplier = {
			value = 0
			if = {
				limit = {
					NOT = { has_law_or_variant = law_type:law_free_trade } # Free traders demand no strait closure for anyone instead
					"strait_access_grievance_against(scope:source_country)" > 0
				}
				value = 1
				add = {
					value = "strait_access_grievance_against(scope:source_country)"
					multiply = 0.07
				}
			}
		}
	}

	wargoal = {
		execution_priority = 21

		contestion_type = control_any_target_country_state

		maneuvers = {
			value = 0

			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 1
			}

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
                    max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy_severity = 0.02
		infamy_min = define:NDiplomacy|WAR_GOAL_INFAMY_MINOR_DEMAND_MIN
		infamy_max = define:NDiplomacy|WAR_GOAL_INFAMY_MINOR_DEMAND_MAX

		infamy = {
			value = 0

			scope:target_country = {
				add = country_infamy_value
			}
		}
	}
}
