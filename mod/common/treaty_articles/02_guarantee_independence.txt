guarantee_independence = {
	kind = directed
	cost = 50

	relations_progress_per_day = 1
	relations_improvement_max = 80

	max_target_involvement = 1000
	target_involvement_applies_to = target_country

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/guarantee_independence_treaties.dds"

	flags = {
		is_guarantee_independence
		can_be_renegotiated
		giftable
		friendly
	}

	maintenance_paid_by = source_country
	usage_limit = once_per_side

	mutual_exclusions = {
		alliance
		defensive_pact
	}

	unlocked_by_technologies = {
		international_relations
	}

	target_modifier = {
		country_treaty_leverage_generation_add = 300
	}

	visible = {
		is_subject = no
	}

	possible = {
		country_rank > scope:other_country.country_rank
		save_temporary_scope_as = actor
		scope:other_country = {
			save_temporary_scope_as = target_country
		}
		custom_tooltip = {
			text = DIPLO_ACTION_RELATIONS_POOR_OR_LESS_NOT
			scope:other_country.relations:root > relations_threshold:poor
		}
	}

	requirement_to_maintain = {
		trigger = {
			scope:source_country.country_rank >= scope:target_country.country_rank
		}
	}

	requirement_to_maintain = {
		trigger = {
			scope:source_country = {
				is_subject = no
			}
		}
	}

	requirement_to_maintain = {
		trigger = {
			scope:target_country = {
				is_subject = no
			}
		}
	}

	can_ratify = {
		scope:source_country = { highest_overlapping_interest_tier = { target = scope:target_country value >= interest_tier_effect_can_sign_guarantee_independence_min_tier } }
		scope:source_country = {
			NOR = {
				custom_tooltip = {
					text = DUPLICATE_ARTICLE_SAME_INPUTS_TREATY
					any_scope_treaty = {
						binds = scope:target_country
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							has_type = guarantee_independence
							source_country = scope:source_country
						}
					}
				}
				custom_description = {
					text = has_treaty_alliance_with_trigger
					subject = this
					object = scope:target_country

					any_scope_treaty = {
						binds = scope:target_country
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							has_type = alliance
						}
					}
				}

				custom_description = {
					text = has_treaty_defensive_pact_with_trigger
					subject = this # Country
					object = scope:target_country # Country

					any_scope_treaty = {
						binds = scope:target_country
						hidden_trigger = {
							OR = {
								scope:treaty = {
									is_renegotiation = no
								}
								scope:treaty.amended_treaty != this
							}
						}
						any_scope_article = {
							has_type = defensive_pact
						}
					}
				}
			}
		}
	}

	on_entry_into_force = {}

	on_withdrawal = {
		scope:withdrawing_country = {
			create_bidirectional_truce = { country = scope:non_withdrawing_country months = 12 }
		}
	}
	on_break = {
		scope:withdrawing_country = {
			change_infamy = {
				value = {
					add = 1
					multiply = {
						value = scope:treaty_options.remaining_binding_period
						min = 365
						divide = 365
					}
					max = 20
					min = 5
				}
			}
		}
		scope:withdrawing_country = {
			change_relations = { country = scope:non_withdrawing_country value = -10 }
		}
	}

	ai = {

		treaty_categories = {
			military_defense
		}

		evaluation_chance = {
			value = 0.0

			if = {
				limit = { country_rank >= rank_value:major_power }
				add = {
					value = country_rank
					multiply = 0.1
				}
			}

			if = {
				limit = {
					any_scope_regional_objective = {
						type = aro:ai_regional_objective_protect_country
						is_active = yes
					}
				}
				add = 0.1
			}
		}

		proposal_weight = {
			value = 1

			if = {
				limit = {
					scope:source_country ?= this
					ai_is_regional_objective_protect_target = { TARGET = scope:target_country }
				}
				add = 3
			}

		# SITAI BEGIN information:guarantee_independence.ai.proposal_weight
		# ACTION program_771a18cc5c508b51 [joint]; signed-envelope consumer action
		if = { limit = { exists = ROOT ROOT = { is_ai = yes } } multiply = {
			value = 1
			# GROUP program_771a18cc5c508b51_treaty_influence_pressure_negative
			add = { value = 0
				# treaty_guarantee_independence_capacity_deficit [diplomacy]; ROOT影响力已为负，升档同一外交容量压力；与低于占用档去重，不把零余额或未知当赤字
				if = { limit = { exists = ROOT ROOT = { is_ai = yes influence < 0.0 } } add = -0.2 }
				# treaty_guarantee_independence_capacity [diplomacy]; 原版每方影响力占用50；ROOT当前余额低于本条款精确占用，仅计边际外交容量压力，不增加资格门
				else_if = { limit = { exists = ROOT ROOT = { is_ai = yes influence < 50.0 } } add = -0.1 }
			}
			# GROUP program_771a18cc5c508b51_treaty_military_financing_pressure_negative
			add = { value = 0
				# treaty_guarantee_independence_default [diplomacy]; ROOT已实际违约，军事条约新增承诺有融资机会成本；不惩罚正常借款，不据此推断任何一方实际付款
				if = { limit = { exists = ROOT ROOT = { is_ai = yes in_default = yes } } add = -0.15 }
			}
			min = 0.65
			max = 1
		} }
		# SITAI END information:guarantee_independence.ai.proposal_weight
	}

		article_ai_usage = {
			offer
			request
		}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					scope:source_country ?= this
				}

				# We are doing this to be able to reuse loc across treaties and diplomatic actions
				scope:source_country = {
					save_temporary_scope_as = target
				}
				scope:target_country = {
					save_temporary_scope_as = actor
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -10
				}

				if = {
					limit = {
						ai_is_regional_objective_protect_target = { TARGET = scope:target_country }
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_REGIONAL_OBJECTIVE_PROTECT"
						value = 25
					}
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ARMY_STRENGTH"
					value = "ai_army_comparison(scope:target_country)"
					multiply = -3
					min = -50
					max = 0
				}

				if = {
					limit = {
						NOT = { is_country_type = unrecognized }
						scope:target_country ?= { is_country_type = unrecognized }
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_UNRECOGNIZED"
						value = -25
					}
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					if = {
						limit = {
							has_attitude = {
								who = scope:target_country
								attitude = protective
							}
						}
						value = 25
					}
					else_if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = antagonistic
								}
								has_attitude = {
									who = scope:target_country
									attitude = belligerent
								}
								has_attitude = {
									who = scope:target_country
									attitude = rebellious
								}
							}
						}
						value = -1000
					}
					else_if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = wary
								}
								has_attitude = {
									who = scope:target_country
									attitude = disinterested
								}
							}
						}
						value = -50
					}
					else_if = {
						limit = {
							has_attitude = {
								who = scope:target_country
								attitude = cautious
							}
						}
						value = -25
					}
				}
			}
			else = {
				scope:source_country = {
					save_temporary_scope_as = actor
				}
				scope:target_country = {
					save_temporary_scope_as = target
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE"
					value = 5
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ARMY_STRENGTH"
					value = "ai_army_comparison(scope:source_country)"
					multiply = -3
					min = 0
					max = 50
				}

				if = {
					limit = {
						scope:target_country.power_bloc ?= {
							NOT = { has_identity = identity:identity_sovereign_empire }
						}
						has_powerful_protectors = {
							ACTOR_SCOPE = scope:source_country
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_POWERFUL_PROTECTORS_THEM"
						value = -25
					}
				}
			}

			# Special anti-Opium Wars cheese - block any alliance/defensive pact/guarantee deals before the JE fires
			if = {
				limit = {
					ai_opium_dealer_no_alliance_trigger = yes
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_OPIUM_DEALER"
					add = -1000
				}
			}

			if = {
				limit = {
					NOT = { exists = scope:is_treaty_active }
					exists = scope:treaty
					scope:treaty.binding_period >= 1825
				}
				multiply = {
					desc = "DIPLOMATIC_ACCEPTANCE_BINDING_PERIOD"
					value = 1
					add = {
						value = scope:treaty.binding_period
						min = 365
						divide = 3650
					}
					max = 2
				}
			}
		}
	}
}
