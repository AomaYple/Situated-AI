ai_strategy_subject_break_free = {
	icon = "gfx/interface/icons/ai_strategy_icons/subject_break_free.dds"

	type = diplomatic

	undesirable_infamy_level = {
		value = 25
	}

	unacceptable_infamy_level = {
		value = 50
	}

	obligation_value = {
		# Rebellious AI wants to use obligations to break free
		if = {
			limit = {
				this.overlord ?= scope:target_country
			}
			add = 50
		}
		else = {
			add = 20
		}
	}

	diplomatic_play_boldness = {
		value = 50
	}

	recklessness = {
		value = 0.5
	}

	aggression = {
		value = 0.5 # Should focus mainly on overlord

		if = {
			limit = {
				is_subject_of = scope:target_country
			}
			add = 10
		}
	}

	building_group_weights = {
		bg_army = 1.5
		bg_military_industry = 1.5
	}

	wargoal_weights = {
		independence = 5.0
		increase_autonomy = 0.5
	}

	secret_goal_scores = {
		antagonize = {
			trigger = {
				is_subject_of = scope:target_country
			}
			score = {
				value = 10000
			}
		}
		none = {
			trigger = {
				is_subject_of = scope:target_country
			}
			score = {
				value = -10000
			}
		}
		defy = {
			trigger = {
				is_subject_of = scope:target_country
			}
			score = {
				value = -10000
			}
		}
		comply = {
			trigger = {
				is_subject_of = scope:target_country
			}
			score = {
				value = -10000
			}
		}
		befriend = {
			trigger = {
				scope:target_country = { country_rank >= root.country_rank }
				can_be_reached_by_target_country = { TARGET_COUNTRY = scope:target_country }
			}
			score = {
				if = {
					limit = {
						scope:target_country = {
							has_diplomatic_pact = {
								who = root.overlord
								type = rivalry
							}
						}
					}
					add = 500
				}
				else_if = {
					limit = {
						scope:target_country = { country_rank >= root.overlord.country_rank }
					}
					add = 100
				}
			}
		}
	}

	become_subject_value = {
		value = -100
	}

	diplomatic_play_support = {
		value = 0

		if = { # Enemies of our overlord
			limit = {
				scope:enemy_country = { root.overlord ?= this }
			}
			add = {
				value = 50
				desc = "AI_STRATEGY_SUPPORT_BREAK_FREE_ENEMY_OF_OVERLORD"
			}
		}
	}

	treaty_category_scores = {
		military_defense = {
			value = 5
		}
	}

	possible = {
		is_subject = yes
		liberty_desire >= 50
	}

	weight = {
		value = liberty_desire
		subtract = 50
		multiply = 2

		# AI implications for Power Blocs: 'Break Free' strategy is more likely for Blocs with Exploitation of Members
		if = {
			limit = {
				power_bloc ?= {
					has_principle_group = principle_group:principle_group_exploit_members
				}
			}
			multiply = 2.0
		}

		# AI implications for Power Blocs: 'Break Free' strategy is more likely for Blocs with Vassalization
		if = {
			limit = {
				power_bloc ?= {
					has_principle_group = principle_group:principle_group_vassalization
				}
			}
			multiply = 1.5
		}

		# Subjects that would collapse if independent should be more loyal
		if = {
			limit = {
				OR = {
					c:DEI ?= this
					has_government_type = gov_chartered_company
				}
			}
			multiply = 0.1
		}

		# Subjects should not start the game wanting independence unless we specifically script this strategy for them
		if = {
			limit = {
				game_date < 1836.1.2
			}
			multiply = 0
		}
	}
}
