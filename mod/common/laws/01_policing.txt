law_local_police = {
	group = lawgroup_policing

	icon = "gfx/interface/icons/law_icons/local_police.dds"

	unlocking_technologies = {
		tech_bureaucracy
	}

	progressiveness = -50

	institution = institution_police
	institution_modifier = {
		interest_group_ig_landowners_pol_str_mult = 0.05
		state_turmoil_effects_mult = -0.05
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_policing_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	# Don't return to local police if we have something more advanced
	ai_will_do = {
		NOR = {
			has_law_or_variant = law_type:law_dedicated_police
			has_law_or_variant = law_type:law_militarized_police
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = {
				has_law_or_variant = law_type:law_no_police
			}
			add = 500
		}

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = {
				has_journal_entry = je_hawaii
			}
			add = 1000
		}

		# SITAI BEGIN information:law_local_police.ai_enact_weight_modifier
		# political_capacity_law_local_police [political]; 合法性极低、存在叛乱性利益集团且人口激进化显著时，为额外立法加入有限政治机会成本；不取消改革、不改变通过率、镇压、法律效果或原版政治候选选择

		# SITAI END information:law_local_police.ai_enact_weight_modifier
	}

	ai_impose_chance = {
		value = 0

		if = {
			limit = {
				has_law = law_type:law_local_police
				power_bloc ?= { has_principle_group = principle_group:principle_group_police_coordination }
			}
			add = base_impose_law_weight
		}

		if = {
			limit = {
				has_law = law_type:law_local_police
				scope:target_country = {
					has_law_or_variant = law_type:law_no_police
				}
				power_bloc ?= { has_principle_group = principle_group:principle_group_police_coordination }
			}
			add = 10
		}

		if = {
			limit = {
				scope:target_country = {
					c:BIC ?= this
					OR = {
						has_journal_entry = je_british_dictate_plantations
						has_journal_entry = je_british_dictate_military
						has_journal_entry = je_british_dictate_universities
					}
				}
			}
			multiply = 0
		}
	}
}
