law_state_religion = {
	group = lawgroup_church_and_state

	icon = "gfx/interface/icons/law_icons/state_religion.dds"

	progressiveness = -50

	is_visible = {
		NOR = { # Turkey should only unlock this as a possibility if it abolishes the Millet System, likewise other Muslim nations
			has_law = law_type:law_millet_system
			has_law = law_type:law_people_of_the_book
		}
		NAND = {
			OR = {
				c:RUS ?= this
				AND = {
					exists = c:RUS
					is_subject_of = c:RUS
					country_is_subject_type_valid_for_caesaropapism = yes
				}
			}
			NOT = {
				has_law_or_variant = law_type:law_theocracy
			}
		}
	}

	on_activate = {
		if = {
			limit = {
				country_has_state_religion = rel:atheist
			}
			set_state_religion = root.country_definition.religion
		}
		if = {
			limit = {
				has_modifier = antireligious_policies
			}
			remove_modifier = antireligious_policies
		}
		recalculate_pop_ig_support = yes
	}

	modifier = {
		interest_group_ig_devout_pol_str_mult = 0.3
		country_authority_add = 200
		country_state_religion_wages_mult = 0.1
		country_non_state_religion_wages_mult = -0.1

		country_acceptance_state_religion_add = 25
		country_acceptance_shared_religious_trait_add = 0
		country_acceptance_shared_religious_trait_group_add = -10
		country_acceptance_no_shared_religious_trait_add = -25

		country_allow_conversion_violent_hostility_bool = yes
		country_allow_conversion_cultural_erasure_bool = yes
		country_allow_conversion_open_prejudice_bool = yes
		country_allow_conversion_second_rate_citizen_bool = yes
		country_allow_conversion_full_acceptance_bool = yes
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_church_and_state_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			has_ideology = ideology:ideology_theocrat
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		# For Japanese religion Journal Entries
		if = {
			limit = {
				OR = {
					has_journal_entry = je_elevate_buddhism
					AND = {
						# If Shinbutsu Bunri is active and the Taikyo Proclamation has been made
						has_journal_entry = je_shinbutsu_bunri
						country_has_state_religion = rel:shinto
					}
				}
			}
			add = 1000
		}

		# SITAI BEGIN information:law_state_religion.ai_enact_weight_modifier
		# ACTION program_f7cf1397de7cf6d3 [joint]; signed-envelope consumer action
		if = { limit = { exists = ROOT ROOT = { is_ai = yes } } add = {
			value = 0
			# GROUP program_f7cf1397de7cf6d3_negative
			add = { value = 0
				# political_capacity_law_state_religion [political]; 合法性极低、存在叛乱性利益集团且人口激进化显著时，为额外立法加入有限政治机会成本；不取消改革、不改变通过率、镇压、法律效果或原版政治候选选择
				if = { limit = { exists = ROOT ROOT = { legitimacy >= 0 legitimacy <= 100 legitimacy < 25 has_insurrectionary_interest_groups = yes radical_fraction = { value >= 0 } radical_fraction = { value <= 1 } radical_fraction = { value >= 0.25 } } } add = -10 }
			}
			# GROUP program_f7cf1397de7cf6d3_law_authority_transition_positive
			add = { value = 0
				# law_context_law_state_religion_law_total_separation_authority_transition [political]; 比较现行到候选的原版权威常数差：余额不足覆盖实际损失时计机会成本，反向恢复只响应已知短缺；不推断人口歧视、移民、技术传播量或税收收益
				if = { limit = { exists = ROOT ROOT = { has_law_or_variant = law_type:law_total_separation authority < 0 } } add = 6 }
				# law_context_law_state_religion_law_freedom_of_conscience_authority_transition [political]; 比较现行到候选的原版权威常数差：余额不足覆盖实际损失时计机会成本，反向恢复只响应已知短缺；不推断人口歧视、移民、技术传播量或税收收益
				else_if = { limit = { exists = ROOT ROOT = { has_law_or_variant = law_type:law_freedom_of_conscience authority < 0 } } add = 6 }
			}
			# GROUP program_f7cf1397de7cf6d3_law_observed_agenda_positive
			add = { value = 0
				# law_context_law_state_religion_agenda_strong [political]; 本国真实政治运动明确支持这个具体法律候选，按已知支持档回应国内诉求；不将政治诉求冒充税收或国力净收益
				if = { limit = { exists = ROOT ROOT = { any_political_movement = { law_enactment_stance = { law = law_type:law_state_religion value > neutral } political_movement_support >= 0.25 } } } add = 10 }
				# law_context_law_state_religion_agenda_base [political]; 本国真实政治运动明确支持这个具体法律候选，按已知支持档回应国内诉求；不将政治诉求冒充税收或国力净收益
				else_if = { limit = { exists = ROOT ROOT = { any_political_movement = { law_enactment_stance = { law = law_type:law_state_religion value > neutral } political_movement_support >= 0.1 } } } add = 6 }
			}

			}

			add = base_impose_law_weight

			if = {
