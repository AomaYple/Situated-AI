sitai_fiscal_entry = {
    OR = {
        in_default = yes
        AND = { taking_loans = yes credit > 0 weeks_until_bankruptcy >= 0 weeks_until_bankruptcy < 26 }
    }
}
sitai_fiscal_hold = {
    OR = {
        in_default = yes
        AND = { taking_loans = yes credit > 0 weeks_until_bankruptcy >= 0 weeks_until_bankruptcy < 52 }
    }
}

    OR = {
        is_at_war = yes
        is_diplomatic_play_initiator = yes
        is_diplomatic_play_target = yes
        is_diplomatic_play_committed_participant = yes
        is_diplomatic_play_undecided_participant = yes
    }
}

    is_ai = yes
