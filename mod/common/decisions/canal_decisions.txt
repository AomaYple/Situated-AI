suez_survey_decision = {
	is_shown = {
		NOT = {
			has_variable = survey_suez_taken
			has_global_variable = suez_canal_complete
		}
		OR = {
			country_rank = rank_value:great_power
			s:STATE_SINAI = {
				any_scope_state = {
					owner = ROOT
				}
			}
		}
		s:STATE_SINAI = {
			NOT = {
				any_scope_state = {
					has_building = building_suez_canal
				}
			}
		}
	}

	possible = {
		has_interest_marker_in_region = sr:region_nile_basin
		has_technology_researched = colonization
		produced_bureaucracy > 1000
	}

	when_taken = {
		set_variable = {
			name = survey_suez_taken
			value = 0
		}
		add_modifier = {
			name = modifier_surveying_suez
		}
		add_journal_entry = {
			type = je_suez_survey
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { produced_bureaucracy > 5000 }
			add = 10
		}
		if = {
			limit = {
				has_strategy = ai_strategy_armed_isolationism
			}
			multiply = 0
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			multiply = 0
		}

suez_canal_purchase = {
	is_shown = {
		has_variable = suez_canal_purchase
		NOT = {
			has_global_variable = suez_canal_purchase_var
			has_global_variable = suez_canal_complete
		}
		s:STATE_SINAI = {
			any_scope_state = {
				NOT = {
					has_building = building_suez_canal
				}
				owner = {
					owns_treaty_port_in = STATE_SINAI
					NOT = {
						has_variable = suez_survey_complete
					}
				}
			}
		}
	}

	possible = {
		gdp > 50000000
		p:xC01060.state.owner = {
			relations:root >= relations_threshold:cordial
			}
		NOT = { has_war_with = ROOT }
	}

	when_taken = {
		set_global_variable = suez_canal_purchase_var
		if = {
			limit = {
				s:STATE_SINAI = {
					any_scope_state = {
						owner = {
							owns_treaty_port_in = STATE_SINAI
							is_player = no
						}
					}
				}
			}
			hidden_effect = {
				s:STATE_SINAI = {
					random_scope_state = {
						limit = {
							owner = {
								owns_treaty_port_in = STATE_SINAI
							}
						}
						owner = {
							add_modifier = {
								name = suez_sale
								months = 120
							}
						}
					}
				}
			}
			s:STATE_SINAI = {
				set_owner_of_provinces = {
					country = ROOT
					provinces = { xC01060 x8ED45E }
				}
			}
			add_modifier = {
				name = suez_purchase
				months = 120
			}
		}
		else = {
			trigger_event = {
				id = canal_events.1
				days = 0
				popup = yes
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { gold_reserves > 15000000 }
			add = 10
		}
		if = {
			limit = {
				net_fixed_income > 100000
				gold_reserves > 0
			}
			add = 10
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			multiply = 0
		}

		# SITAI BEGIN information:suez_canal_purchase.ai_chance
		# ACTION program_a60ba8d4e9ee1cc9 [joint]; signed-envelope consumer action
		if = { limit = { exists = ROOT ROOT = { is_ai = yes } } multiply = {
			value = 1
			# GROUP program_a60ba8d4e9ee1cc9_canal_purchase_cashflow_negative
